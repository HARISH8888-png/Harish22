(function(){
  var reduce = matchMedia('(prefers-reduced-motion:reduce)').matches;
  var intro = document.getElementById('intro'), done = false;

  /* ---- Opening (plays once, under 2.5s, skippable) ---- */
  function endIntro(){ if(done) return; done = true; intro.classList.add('gone'); startRoles(); }
  if(reduce){ endIntro(); }
  else{
    setTimeout(function(){ intro.classList.add('go'); }, 650);
    setTimeout(endIntro, 2250);
    document.getElementById('skip').addEventListener('click', endIntro);
  }

  /* ---- Signature move: smooth role cycling ---- */
  var ul = document.getElementById('roles'), n = ul.children.length - 1, i = 0, timer;
  function step(){
    i++; ul.style.transition = ''; ul.style.transform = 'translateY(' + (-i * 1.1) + 'em)';
    if(i === n){
      setTimeout(function(){ ul.style.transition = 'none'; ul.style.transform = 'translateY(0)'; i = 0; }, 750);
    }
  }
  function startRoles(){ if(!reduce && !timer) timer = setInterval(step, 2200); }

  /* ---- Collapse / restore windows ---- */
  document.querySelectorAll('.mb').forEach(function(b){
    b.addEventListener('click', function(){ b.closest('.win').classList.toggle('min'); });
  });
  document.querySelectorAll('a[href^="#"]').forEach(function(a){
    a.addEventListener('click', function(){
      var t = document.querySelector(a.getAttribute('href'));
      if(t) t.classList.remove('min');
    });
  });

  var links = {}; document.querySelectorAll('.tasks a').forEach(function(a){ links[a.hash.slice(1)] = a; });
  if('IntersectionObserver' in window){
    var io = new IntersectionObserver(function(es){
      es.forEach(function(e){ if(e.isIntersecting){
        Object.keys(links).forEach(function(k){ links[k].classList.toggle('on', k === e.target.id); });
        if(links[e.target.id]) links[e.target.id].scrollIntoView({inline:'center',block:'nearest'});
      }});
    }, {rootMargin:'-35% 0px -55% 0px'});
    Object.keys(links).forEach(function(k){ io.observe(document.getElementById(k)); });
  }
  var clock = document.getElementById('clock');
  function tick(){ clock.textContent = new Date().toLocaleTimeString([], {hour:'2-digit', minute:'2-digit', hour12:false}); }
  tick(); setInterval(tick, 20000);

  /* ---- Easter egg: click the clock five times ---- */
  var taps = 0, tapT, egg = document.getElementById('egg');
  clock.addEventListener('click', function(){
    taps++; clearTimeout(tapT); tapT = setTimeout(function(){ taps = 0; }, 1500);
    if(taps >= 5){ taps = 0; egg.classList.add('show'); document.getElementById('eggok').focus(); }
  });
  document.getElementById('eggok').addEventListener('click', function(){ egg.classList.remove('show'); });
  document.addEventListener('keydown', function(e){ if(e.key === 'Escape') egg.classList.remove('show'); });
